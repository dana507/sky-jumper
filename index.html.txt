  canvas { width:100%; height:100%; display:block; background:linear-gradient(#87CEEB,#E0F7FF); border-radius:12px; }
  #ui { position:absolute; inset:0; pointer-events:none; color:#fff; text-shadow:0 2px 4px rgba(0,0,0,.5); }
  .top { position:absolute; top:12px; right:12px; left:12px; display:flex; justify-content:space-between; font-weight:700; }
  .center { position:absolute; inset:0; display:flex; flex-direction:column; justify-content:center; align-items:center; gap:12px; background:rgba(0,0,0,.45); pointer-events:auto; }
  button { pointer-events:auto; border:0; border-radius:12px; padding:12px 18px; font-size:16px; font-weight:700; cursor:pointer; background:#ffd166; color:#222; }

<div id="game-wrap">
  <canvas id="c" width="360" height="640"></canvas>
  <div id="ui">
    <div class="top">
      <div>النقاط: <span id="score">0</span></div>
      <div>العملات: <span id="coins">0</span></div>
    </div>
    <div id="overlay" class="center hidden">
      <h2 id="title">انتهت اللعبة</h2>
      <div>أفضل نتيجة: <span id="best">0</span></div>
      <button id="reviveBtn" class="sec">شاهد إعلانًا للمتابعة</button>
      <button id="restartBtn">إعادة</button>
      <button id="skinBtn">لون جديد (50 عملة)</button>
    </div>
  </div>
</div>
<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');
const W = canvas.width, H = canvas.height;

const ui = {
  score: document.getElementById('score'),
  coins: document.getElementById('coins'),
  best: document.getElementById('best'),
  overlay: document.getElementById('overlay'),
  title: document.getElementById('title'),
  reviveBtn: document.getElementById('reviveBtn'),
  restartBtn: document.getElementById('restartBtn'),
  skinBtn: document.getElementById('skinBtn')
};

let state = {
  running: false,
  gameOver: false,
  score: 0,
  coins: parseInt(localStorage.getItem('sky_coins') || '0'),
  best: parseInt(localStorage.getItem('sky_best') || '0'),
  revived: false,
  gameOverCount: 0,
  player: { x: W/2 - 15, y: H - 100, w: 30, h: 30, vy: 0, color: '#ff6b6b' },
  platforms: [],
  coinsList: [],
  keys: { left:false, right:false },
  pointerX: null
};

function reset() {
  state.running = true;
  state.gameOver = false;
  state.score = 0;
  state.revived = false;
  state.player.x = W/2 - 15;
  state.player.y = H - 100;
  state.player.vy = -10;
  state.platforms = [];
  state.coinsList = [];
  state.platforms.push({x: W/2 - 40, y: H - 60, w: 80, h: 12});
  for (let i=0; i<10; i++) addPlatform();
  ui.overlay.classList.add('hidden');
  updateUI();
}

function addPlatform() {
  const minY = state.platforms.length ? Math.min(...state.platforms.map(p=>p.y)) : H;
  const y = minY - (60 + Math.random()*40);
  const w = 60 + Math.random()*40;
  const x = Math.random() * (W - w);
  state.platforms.push({x, y, w, h:12});
  if (Math.random() < 0.35) {
    state.coinsList.push({x: x + w/2, y: y - 25, r: 8, taken:false});
  }
}

function updateUI() {
  ui.score.textContent = Math.floor(state.score);
  ui.coins.textContent = state.coins;
  ui.best.textContent = state.best;
}

function showRewardedAd(callback) {
  console.log('Rewarded ad placeholder');
  setTimeout(callback, 600);
}

function showInterstitial() {
  console.log('Interstitial ad placeholder');
}

function gameOver() {
  state.running = false;
  state.gameOver = true;
  state.gameOverCount++;
  if (state.score > state.best) {
    state.best = Math.floor(state.score);
    localStorage.setItem('sky_best', state.best);
  }
  ui.title.textContent = 'انتهت اللعبة';
  ui.reviveBtn.classList.toggle('hidden', state.revived);
  ui.overlay.classList.remove('hidden');
  updateUI();
  if (state.gameOverCount % 3 === 0) showInterstitial();
}

ui.reviveBtn.onclick = () => {
  showRewardedAd(() => {
    state.revived = true;
    state.gameOver = false;
    state.running = true;
    state.player.y = H - 150;
    state.player.vy = -10;
    state.platforms = state.platforms.filter(p => p.y < state.player.y - 50);
    state.platforms.push({x: state.player.x - 20, y: state.player.y + 60, w: 80, h: 12});
    while (state.platforms.length < 8) addPlatform();
    ui.overlay.classList.add('hidden');
  });
};

ui.restartBtn.onclick = () => reset();
ui.skinBtn.onclick = () => {
  if (state.coins >= 50) {
    state.coins -= 50;
    localStorage.setItem('sky_coins', state.coins);
    const colors = ['#ff6b6b','#4ecdc4','#ffe66d','#a06cd5','#06d6a0','#ff9f1c'];
    state.player.color = colors[Math.floor(Math.random()*colors.length)];
    updateUI();
  } else {
    alert('تحتاج 50 عملة');
  }
};

canvas.addEventListener('pointerdown', e => {
  const rect = canvas.getBoundingClientRect();
  state.pointerX = (e.clientX - rect.left) * (W / rect.width);
});
canvas.addEventListener('pointermove', e => {
  if (state.pointerX !== null) {
    const rect = canvas.getBoundingClientRect();
    state.pointerX = (e.clientX - rect.left) * (W / rect.width);
  }
});
canvas.addEventListener('pointerup', () => { state.pointerX = null; });
window.addEventListener('keydown', e => {
  if (e.key === 'ArrowLeft') state.keys.left = true;
  if (e.key === 'ArrowRight') state.keys.right = true;
  if (e.key === ' ') { if (state.gameOver) reset(); }
});
window.addEventListener('keyup', e => {
  if (e.key === 'ArrowLeft') state.keys.left = false;
  if (e.key === 'ArrowRight') state.keys.right = false;
});

function update() {
  if (!state.running) return;
  const p = state.player;
  const prevBottom = p.y + p.h;
  p.vy += 0.45;
  p.y += p.vy;

  if (state.pointerX !== null) {
    p.x += (state.pointerX - (p.x + p.w/2)) * 0.18;
  }
  if (state.keys.left) p.x -= 5;
  if (state.keys.right) p.x += 5;
  p.x = Math.max(0, Math.min(W - p.w, p.x));

  if (p.vy > 0) {
    for (const plat of state.platforms) {
      if (prevBottom <= plat.y && p.y + p.h >= plat.y &&
          p.x + p.w > plat.x && p.x < plat.x + plat.w) {
        p.y = plat.y - p.h;
        p.vy = -10.5;
        break;
      }
    }
  }

  for (const c of state.coinsList) {
    if (!c.taken && Math.hypot((p.x+p.w/2)-c.x, (p.y+p.h/2)-c.y) < c.r + 15) {
      c.taken = true;
      state.coins++;
      localStorage.setItem('sky_coins', state.coins);
    }
  }

  if (p.y < H/2) {
    const shift = H/2 - p.y;
    p.y = H/2;
    state.score += shift * 0.1;
    for (const plat of state.platforms) plat.y += shift;
    for (const c of state.coinsList) c.y += shift;
    state.platforms = state.platforms.filter(pl => pl.y < H + 50);
    state.coinsList = state.coinsList.filter(c => c.y < H + 50 && !c.taken);
    while (state.platforms.length < 10) addPlatform();
  }

  if (p.y > H + 60) gameOver();
  updateUI();
}

function draw() {
  ctx.clearRect(0,0,W,H);
  ctx.fillStyle = 'rgba(255,255,255,.6)';
  for (let i=0; i<5; i++) {
    const x = (i*97 + state.score*0.2) % (W+100) - 50;
    const y = 80 + i*110;
    ctx.beginPath();
    ctx.arc(x, y, 30, 0, Math.PI*2);
    ctx.arc(x+30, y+5, 22, 0, Math.PI*2);
    ctx.arc(x-25, y+8, 20, 0, Math.PI*2);
    ctx.fill();
  }
  for (const plat of state.platforms) {
    ctx.fillStyle = '#2a9d8f';
    ctx.fillRect(plat.x, plat.y, plat.w, plat.h);
    ctx.fillStyle = '#21867a';
    ctx.fillRect(plat.x, plat.y+8, plat.w, 4);
  }
  for (const c of state.coinsList) {
    if (c.taken) continue;
    ctx.fillStyle = '#ffd166';
    ctx.beginPath();
    ctx.arc(c.x, c.y, c.r, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#f4a261';
    ctx.beginPath();
    ctx.arc(c.x, c.y, c.r*0.6, 0, Math.PI*2);
    ctx.fill();
  }
  const p = state.player;
  ctx.fillStyle = p.color;
  ctx.fillRect(p.x, p.y, p.w, p.h);
  ctx.fillStyle = '#fff';
  ctx.beginPath();
  ctx.arc(p.x+10, p.y+11, 4, 0, Math.PI*2);
  ctx.arc(p.x+20, p.y+11, 4, 0, Math.PI*2);
  ctx.fill();
  ctx.fillStyle = '#222';
  ctx.beginPath();
  ctx.arc(p.x+10, p.y+11, 2, 0, Math.PI*2);
  ctx.arc(p.x+20, p.y+11, 2, 0, Math.PI*2);
  ctx.fill();
}

let last = performance.now();
function loop(now) {
  const dt = Math.min(32, now - last) / 16.67;
  last = now;
  update(dt);
  draw();
  requestAnimationFrame(loop);
}

reset();
requestAnimationFrame(loop);
</script>
